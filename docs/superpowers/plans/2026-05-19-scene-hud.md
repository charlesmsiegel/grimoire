    ids_seen: set[str] = set()

Small recursive-descent parser; no eval. Tokenizer + AST + evaluator. Limit grammar exactly as spec:

```python
# backend/src/grimoire/hud/expression.py
from __future__ import annotations

import logging
import re
from dataclasses import dataclass, field
from typing import Callable

log = logging.getLogger(__name__)


class ParseError(Exception):
    pass


@dataclass
class Token:
    kind: str        # IDENT | DOT | LPAREN | RPAREN | COMMA | EQ | STRING | INT | BOOL | AND | OR | NOT
    value: object


_TOKEN_RE = re.compile(
    r"""
    \s+
    | (?P<STRING>"(?:[^"\\]|\\.)*")
    | (?P<INT>\d+)
    | (?P<BOOL>\btrue\b|\bfalse\b)
    | (?P<AND>\band\b)
    | (?P<OR>\bor\b)
    | (?P<NOT>\bnot\b)
    | (?P<IDENT>[a-zA-Z_][a-zA-Z_0-9]*)
    | (?P<DOT>\.)
    | (?P<LPAREN>\()
    | (?P<RPAREN>\))
    | (?P<COMMA>,)
    | (?P<EQ>=)
    """,
    re.VERBOSE,
)


def _tokenize(s: str) -> list[Token]:
    pos = 0
    toks: list[Token] = []
    while pos < len(s):
        m = _TOKEN_RE.match(s, pos)
        if m is None or m.group(0).isspace() is False and m.lastgroup is None:
            raise ParseError(f"unexpected char at {pos}: {s[pos]!r}")
        if m.group(0).strip() == "":
            pos = m.end()
            continue
        kind = m.lastgroup
        value = m.group(kind)
        if kind == "STRING":
            value = value[1:-1].encode().decode("unicode_escape")
        elif kind == "INT":
            value = int(value)
        elif kind == "BOOL":
            value = value == "true"
        toks.append(Token(kind, value))
        pos = m.end()
    return toks


@dataclass
class Expr:
    pass


@dataclass
class Var(Expr):
    path: tuple[str, ...]


@dataclass
class Call(Expr):
    path: tuple[str, ...]
    args: dict[str, object]


@dataclass
class Not(Expr):
    inner: Expr


@dataclass
class And(Expr):
    left: Expr
    right: Expr


@dataclass
class Or(Expr):
    left: Expr
    right: Expr


@dataclass
class BoolLit(Expr):
    value: bool


# Parser implementation: standard recursive descent
class _Parser:
    def __init__(self, toks: list[Token]):
        self.toks = toks
        self.i = 0

    def _peek(self) -> Token | None:
        return self.toks[self.i] if self.i < len(self.toks) else None

    def _consume(self, kind: str | None = None) -> Token:
        t = self.toks[self.i]
        if kind and t.kind != kind:
            raise ParseError(f"expected {kind}, got {t.kind}")
        self.i += 1
        return t

    def parse(self) -> Expr:
        e = self._or()
        if self._peek() is not None:
            raise ParseError(f"trailing tokens: {self.toks[self.i:]}")
        return e

    def _or(self) -> Expr:
        left = self._and()
        while self._peek() and self._peek().kind == "OR":
            self._consume("OR")
            left = Or(left, self._and())
        return left

    def _and(self) -> Expr:
        left = self._atom()
        while self._peek() and self._peek().kind == "AND":
            self._consume("AND")
            left = And(left, self._atom())
        return left

    def _atom(self) -> Expr:
        t = self._peek()
        if t is None:
            raise ParseError("expected expression")
        if t.kind == "NOT":
            self._consume("NOT")
            return Not(self._atom())
        if t.kind == "BOOL":
            self._consume("BOOL")
            return BoolLit(t.value)
        # Var or Call (path)
        path = [self._consume("IDENT").value]
        while self._peek() and self._peek().kind == "DOT":
            self._consume("DOT")
            path.append(self._consume("IDENT").value)
        if self._peek() and self._peek().kind == "LPAREN":
            self._consume("LPAREN")
            args: dict[str, object] = {}
            if self._peek() and self._peek().kind != "RPAREN":
                while True:
                    key = self._consume("IDENT").value
                    self._consume("EQ")
                    v_tok = self._consume()
                    args[key] = v_tok.value
                    if self._peek() and self._peek().kind == "COMMA":
                        self._consume("COMMA")
                        continue
                    break
            self._consume("RPAREN")
            return Call(tuple(path), args)
        return Var(tuple(path))


def parse_expression(s: str) -> Expr:
    return _Parser(_tokenize(s)).parse()


@dataclass
class EvaluationContext:
    scene: dict = field(default_factory=dict)
    pc: dict = field(default_factory=dict)
    mechanics: dict = field(default_factory=dict)
    present_npc: dict = field(default_factory=dict)


def evaluate(s: str | Expr, ctx: EvaluationContext) -> bool:
    expr = parse_expression(s) if isinstance(s, str) else s
    try:
        return bool(_eval(expr, ctx))
    except Exception as e:
        log.warning("expression evaluation failed: %s", e)
        return False


def _eval(expr: Expr, ctx: EvaluationContext) -> object:
    if isinstance(expr, BoolLit):
        return expr.value
    if isinstance(expr, Var):
        return _resolve_path(expr.path, ctx)
    if isinstance(expr, Call):
        target = _resolve_path(expr.path, ctx)
        if not callable(target):
            raise ParseError(f"call on non-callable: {expr.path}")
        return target(**expr.args)
    if isinstance(expr, Not):
        return not _eval(expr.inner, ctx)
    if isinstance(expr, And):
        return bool(_eval(expr.left, ctx)) and bool(_eval(expr.right, ctx))
    if isinstance(expr, Or):
        return bool(_eval(expr.left, ctx)) or bool(_eval(expr.right, ctx))
    raise ParseError(f"unknown expr: {expr}")


def _resolve_path(path: tuple[str, ...], ctx: EvaluationContext) -> object:
    root = {
        "scene": ctx.scene, "pc": ctx.pc, "mechanics": ctx.mechanics, "present_npc": ctx.present_npc,
    }
    cur = root.get(path[0])
    if cur is None:
        raise KeyError(path[0])
    for seg in path[1:]:
        if isinstance(cur, dict):
            cur = cur[seg]
        else:
            cur = getattr(cur, seg)
    return cur
```

```python
# backend/src/grimoire/hud/service.py
import asyncio
from dataclasses import dataclass

import httpx

from grimoire.hud.expression import EvaluationContext, evaluate
from grimoire.hud.widgets import CORE_WIDGETS
from grimoire.types.hud import HudWidget


@dataclass(frozen=True, slots=True)
class WidgetSnapshot:
    id: str
    status: str         # ok | error | timeout | hidden
    data: object | None
    error: str | None = None
    stale: bool = False


@dataclass(frozen=True, slots=True)
class AggregateResult:
    campaign_id: str
    scene_id: str | None
    widgets: list[WidgetSnapshot]


class HudService:
    def __init__(self, *, http: httpx.AsyncClient, config, mechanics_loader, scene_manager, characters_service):
        self.http = http
        self.config = config
        self.mechanics = mechanics_loader
        self.scenes = scene_manager
        self.characters = characters_service

    async def aggregate(self, campaign_id: str, *, observer=None) -> AggregateResult:
        scene = await self.scenes.current_for_campaign(campaign_id)
        ctx = await self._eval_context(campaign_id, scene)
        widgets = list(CORE_WIDGETS) + await self._mechanics_widgets(campaign_id)
        config = await self._load_config(campaign_id)
        visible_widgets = []
        for w in widgets:
            if not config.widget_visible(w.id):
                continue
            if w.visible_when and not evaluate(w.visible_when, ctx):
                continue
            visible_widgets.append(w)
        results = await asyncio.gather(*[self._fetch_widget(w, campaign_id, scene, observer)
                                         for w in visible_widgets])
        return AggregateResult(campaign_id=campaign_id,
                               scene_id=scene.id if scene else None,
                               widgets=results)

    async def _fetch_widget(self, w: HudWidget, campaign_id, scene, observer) -> WidgetSnapshot:
        endpoint = w.read.endpoint.replace("{sid}", scene.id if scene else "")
        url = f"http://localhost{endpoint}".replace("/campaigns/{id}", f"/campaigns/{campaign_id}")
        # Actually use in-process dispatch via FastAPI app testclient or direct service call.
        # Simpler: route owner-module → direct service call (no HTTP) — keeps latency tight.
        try:
            data = await asyncio.wait_for(
                self._call_owner_direct(w, campaign_id, scene, observer),
                timeout=self.config.aggregate_timeout_seconds_per_widget,
            )
            return WidgetSnapshot(id=w.id, status="ok", data=data)
        except asyncio.TimeoutError:
            return WidgetSnapshot(id=w.id, status="timeout", data=None,
                                  error="owner endpoint timeout")
        except Exception as e:
            return WidgetSnapshot(id=w.id, status="error", data=None, error=str(e))

    async def _call_owner_direct(self, w, campaign_id, scene, observer):
        # Maps widget id → owner method. Implementation per widget.
        if w.id == "core.in-game-date":
            return await self.time_engine.current_date(campaign_id)
        if w.id == "core.present-cast":
            return await self.characters.present_cast_payload(campaign_id, scene, observer)
        # ... etc.
        raise NotImplementedError(w.id)

    async def _eval_context(self, campaign_id, scene) -> EvaluationContext:
        # Build the scene/pc/mechanics dicts for visible_when evaluation
        return EvaluationContext(
            scene={"combat_active": scene and scene.combat_active or False,
                   "has_present": lambda role: any(p.role == role for p in scene.present)},
            pc={"has_sheet": ..., "is_active": ..., "in_scene": ...},
            mechanics={"has_event": ..., "is_active": ...},
            present_npc={"has_sheet": ..., "count": ...},
        )

    async def _mechanics_widgets(self, campaign_id) -> list[HudWidget]:
        active = await self.mechanics.active_for_campaign(campaign_id)
        return active.manifest.hud_widgets if active else []
```
