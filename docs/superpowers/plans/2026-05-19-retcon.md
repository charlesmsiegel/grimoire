# Retcon Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development or superpowers:executing-plans. Steps use checkbox (`- [ ]`) syntax.

**Goal:** Land everything in `docs/superpowers/specs/2026-05-19-retcon-design.md`. Adds the replay variant of retcon on top of the existing leave-as-is variant. Hard dep on `swipes-alternates` (replay produces alternates via `regenerate_post` + `switch_primary_alternate`).

**Architecture:** Four sequential branches.

- **A** `feature/retcon-A-replay-result-types` — extend `RetconResult`; add EventType entries; backend serialization unchanged for leave path.
- **B** `feature/retcon-B-batch-session` — `RetconReplaySession` state machine: per-post regenerate, accept, try-again, cancel.
- **C** `feature/retcon-C-rest-ws` — REST routes + WS events.
- **D** `feature/retcon-D-frontend` — Full-screen replay modal + contradiction badges + fork nudge.

**Tech Stack:** Python 3.12, FastAPI, pytest-asyncio, Pydantic v2; React/TS frontend.

```python
# backend/src/grimoire/orchestrator/retcon_replay.py
from __future__ import annotations

from dataclasses import dataclass, field
from typing import Optional

from grimoire.types.orchestrator import EventType, RetconResult


@dataclass(slots=True)
class ReplayBatchState:
    batch_id: str
    campaign_id: str
    edited_post_id: str
    subsequent_post_ids: list[str]
    current_index: int = 0
    current_alternate_id: str | None = None
    accepted_post_ids: list[str] = field(default_factory=list)
    contradictions: list[str] = field(default_factory=list)
    completed: bool = False
    cancelled_at_post_id: str | None = None


class RetconReplaySession:
    def __init__(self, orchestrator, store, scene_manager, continuity, events):
        self.orch = orchestrator
        self.store = store
        self.scenes = scene_manager
        self.continuity = continuity
        self.events = events
        self._batches: dict[str, ReplayBatchState] = {}        # campaign_id -> state

    def is_active(self, campaign_id: str) -> bool:
        b = self._batches.get(campaign_id)
        return b is not None and not b.completed

    async def start(self, *, campaign_id: str, edited_post_id: str) -> ReplayBatchState:
        if self.is_active(campaign_id):
            raise RetconInFlight(campaign_id)
        batch_id = f"rb_{uuid7()}"
        subsequent = await self._collect_subsequent_post_ids(campaign_id, edited_post_id)
        state = ReplayBatchState(batch_id=batch_id, campaign_id=campaign_id,
                                 edited_post_id=edited_post_id,
                                 subsequent_post_ids=subsequent)
        self._batches[campaign_id] = state
        await self.events.emit(campaign_id, EventType.RETCON_STARTED, {
            "post_id": edited_post_id, "replay_subsequent": True, "batch_id": batch_id,
        })
        if subsequent:
            await self._generate_alternate_for_current(state)
        else:
            state.completed = True
            await self.events.emit(campaign_id, EventType.RETCON_COMPLETE, {"batch_id": batch_id})
        return state

    async def accept(self, campaign_id: str) -> ReplayBatchState:
        state = self._batches[campaign_id]
        if state.completed:
            raise RetconBatchClosed()
        post_id = state.subsequent_post_ids[state.current_index]
        alt_id = state.current_alternate_id
        await self.orch.switch_primary_alternate(campaign_id=campaign_id,
                                                 post_id=post_id, alternate_id=alt_id)
        contras = await self.continuity.check_for_post(post_id)
        state.contradictions.extend(c.id for c in contras)
        state.accepted_post_ids.append(post_id)
        await self.events.emit(campaign_id, EventType.RETCON_POST_ACCEPTED, {
            "post_id": post_id, "alternate_id": alt_id, "batch_id": state.batch_id,
        })
        state.current_index += 1
        if state.current_index >= len(state.subsequent_post_ids):
            state.completed = True
            await self.events.emit(campaign_id, EventType.RETCON_COMPLETE, {"batch_id": state.batch_id})
        else:
            await self._generate_alternate_for_current(state)
        return state

    async def try_again(self, campaign_id: str) -> ReplayBatchState:
        state = self._batches[campaign_id]
        post_id = state.subsequent_post_ids[state.current_index]
        # Drop the current alternate (cleanup deltas)
        if state.current_alternate_id:
            await self.orch.delete_alternate(post_id=post_id, alternate_id=state.current_alternate_id)
        await self._generate_alternate_for_current(state)
        return state

    async def cancel(self, campaign_id: str) -> ReplayBatchState:
        state = self._batches[campaign_id]
        post_id_at_cancel = state.subsequent_post_ids[state.current_index]
        if state.current_alternate_id:
            await self.orch.delete_alternate(
                post_id=post_id_at_cancel, alternate_id=state.current_alternate_id,
            )
        state.cancelled_at_post_id = post_id_at_cancel
        state.completed = True
        await self.events.emit(campaign_id, EventType.RETCON_CANCELLED, {
            "batch_id": state.batch_id, "cancelled_at_post_id": post_id_at_cancel,
        })
        return state

    def get(self, campaign_id: str, batch_id: str) -> ReplayBatchState:
        s = self._batches.get(campaign_id)
        if s is None or s.batch_id != batch_id:
            raise RetconBatchNotFound(batch_id)
        return s

    async def _generate_alternate_for_current(self, state: ReplayBatchState) -> None:
        post_id = state.subsequent_post_ids[state.current_index]
        regen = await self.orch.regenerate_post(
            campaign_id=state.campaign_id, post_id=post_id,

        await self.scenes.tag_alternate_with_batch(post_id, regen.new_alternate_id, state.batch_id)

            "post_id": post_id, "new_alternate_id": regen.new_alternate_id,
            "batch_id": state.batch_id,
```

```python
# backend/src/grimoire/orchestrator/service.py
async def retcon_post(
    self,
    *,
    campaign_id: str,
    post_id: str,
    new_text: str,
    replay_subsequent: bool = False,
) -> RetconResult:
    # Existing leave-as-is path runs first to actually edit the post text
    base = await self._retcon_leave_as_is(campaign_id, post_id, new_text)

    if not replay_subsequent:
        return base

    state = await self.retcon_replay.start(campaign_id=campaign_id, edited_post_id=post_id)
    return replace(base,
        replay_batch_id=state.batch_id,
        replayed_post_ids=[],     # populated as user accepts
        contradictions_detected=[],
    )
```
