# Fork (Campaign-Level) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development or superpowers:executing-plans.

**Goal:** Land everything in `docs/superpowers/specs/2026-05-19-fork-design.md`. Campaign-level fork (full new campaign id), image hardlink probe + auto-fallback, audit-log replay for fork-from-earlier, lineage tree. Existing branch-level fork in `orchestrator.fork_branch` stays.

**Architecture:** Five branches.

- **A** `feature/fork-A-schema` — `campaigns` provenance columns + `pending_forks` queue table.
- **B** `feature/fork-B-bulk-copy` — `state_store/fork.py` with `bulk_copy`, `fingerprint_at_turn`, `replay_to_turn`.
- **C** `feature/fork-C-images` — hardlink probe + auto-fallback module.
- **D** `feature/fork-D-orchestrator` — `fork_campaign` entry point with queueing for mid-stream.
- **E** `feature/fork-E-rest-frontend` — REST routes (forks + lineage), fork dialog, indented campaign list.

**Tech Stack:** Python 3.12, FastAPI, aiosqlite, pytest-asyncio, Pydantic v2; React/TS frontend; PowerShell shell on Windows.

async def bulk_copy(
    store: StateStore,
    *,
    original: str,
    new: str,
    cutoff_turn_no: int | None,
) -> None:
    async with store.db.connect_write() as conn:
        async with conn.transaction():
            for table, turn_col in CAMPAIGN_SCOPED_TABLES:
                if cutoff_turn_no is not None and turn_col:
                    sql = (
                        f"INSERT INTO {table} SELECT * FROM {table} "
                        f"WHERE campaign_id=? AND {turn_col} <= ?"
                    )
                    params = (original, cutoff_turn_no)
                else:
                    sql = f"INSERT INTO {table} SELECT * FROM {table} WHERE campaign_id=?"
                    params = (original,)
                # Replace campaign_id in the inserted rows
                # SQLite doesn't support column-rename in INSERT..SELECT directly; use a temp form
                # via explicit column list. Implementation detail — adjust per table schema.
                await _execute_with_id_rewrite(conn, table, original, new, turn_col, cutoff_turn_no)


async def fingerprint_at_turn(store: StateStore, campaign_id: str, turn_id: str) -> str:
    """Stable hash of all campaign-scoped state at the given turn.

    Computed over a canonical serialization of the key tables; SHA-256.
    """
    h = hashlib.sha256()
    async with store.db.connect_read() as conn:
        for table, turn_col in CAMPAIGN_SCOPED_TABLES:
            rows = await conn.execute_fetchall(
                f"SELECT * FROM {table} WHERE campaign_id=? ORDER BY rowid",
                (campaign_id,),
            )
            for r in rows:
                h.update(table.encode())
                h.update(json.dumps(dict(r), sort_keys=True, default=str).encode())
    return h.hexdigest()


async def replay_to_turn(
    store: StateStore,
    *,
    original: str,
    new: str,
    cutoff: str,
) -> int:
    """Replay deltas from `original` campaign into `new` up to cutoff."""
    async with store.db.connect_write() as conn:
        rows = await conn.execute_fetchall(
            "SELECT * FROM deltas WHERE campaign_id=? AND turn_no <= ? ORDER BY applied_at ASC",
            (original, cutoff_turn_no_for(cutoff)),
        )
    count = 0
    async with store.db.connect_write() as conn:
        async with conn.transaction():
            for r in rows:
                await store.apply_delta_from_row(conn, r, campaign_id=new)
                count += 1
    return count
```

# backend/src/grimoire/orchestrator/fork_images.py
import asyncio
import os
import shutil
from dataclasses import dataclass
from pathlib import Path


@dataclass(frozen=True, slots=True)
class ImageHandlingResult:
    handling: str       # hardlink | deep_copy | mixed
    files_copied: int


async def fork_image_files(original_dir: Path, new_dir: Path) -> ImageHandlingResult:
    src_images = original_dir / "images"
    dst_images = new_dir / "images"
    dst_images.mkdir(parents=True, exist_ok=True)
    if not src_images.exists():
        return ImageHandlingResult("hardlink", 0)

    # Probe
    sentinel_src = src_images / ".sentinel-fork-probe"
    sentinel_dst = dst_images / ".sentinel-fork-probe"
    try:
        sentinel_src.touch()
        os.link(sentinel_src, sentinel_dst)
        handling = "hardlink"
    except (OSError, PermissionError):
        handling = "deep_copy"
    finally:
        sentinel_src.unlink(missing_ok=True)
        sentinel_dst.unlink(missing_ok=True)

    count = 0
    downgraded = False
    for src in src_images.rglob("*"):
        if src.is_dir() or src.name.startswith(".sentinel"):
            continue
        dst = dst_images / src.relative_to(src_images)
        dst.parent.mkdir(parents=True, exist_ok=True)
        if handling == "hardlink":
            try:
                os.link(src, dst)
            except OSError:
                shutil.copy2(src, dst)
                downgraded = True
        else:
            shutil.copy2(src, dst)
        count += 1

    if downgraded:
        handling = "mixed"
    return ImageHandlingResult(handling=handling, files_copied=count)
```

```python
async def fork_campaign(
    self,
    *,
    campaign_id: str,
    new_campaign_id: str,
    new_name: str,
    fork_at_post_id: str | None = None,
    description: str | None = None,
    make_active: bool = False,
) -> ForkCampaignResult:
    # Validate
    if await self.store.campaign_exists(new_campaign_id):
        raise CampaignIdExists(new_campaign_id)

    # Queue if streaming
    if self._is_streaming.get(campaign_id):
        return await self._enqueue_fork(
            campaign_id, new_campaign_id, new_name, fork_at_post_id,
            description, make_active,
        )

    return await self._execute_fork(
        campaign_id=campaign_id, new_campaign_id=new_campaign_id,
        new_name=new_name, fork_at_post_id=fork_at_post_id,
        description=description, make_active=make_active,
    )


async def _execute_fork(self, ...) -> ForkCampaignResult:
    await self.events.emit_campaign_fork_started(campaign_id, new_campaign_id, fork_at_post_id)

    # 1. Create the campaign row + directory
    await self.store.create_campaign(
        new_campaign_id, new_name, setting_id=..., forked_from_campaign_id=campaign_id,
        forked_at_post_id=fork_at_post_id,
    )
    new_dir = self.store.paths.campaign_root(new_campaign_id)
    new_dir.mkdir(parents=True, exist_ok=True)

    # 2. Copy narrative files
    src_dir = self.store.paths.campaign_root(campaign_id)
    await self._copy_narrative_files(src_dir, new_dir, cutoff_post=fork_at_post_id)

    # 3. Bulk-copy SQLite (or replay for fork-from-earlier)
    deltas_replayed = 0
    fingerprint_match = True
    degraded = False
    if fork_at_post_id is None:
        await bulk_copy(self.store, original=campaign_id, new=new_campaign_id, cutoff_turn_no=None)
    else:
        cutoff_turn_no = await self.store.turn_no_of_post(fork_at_post_id)
        fp_origin = await fingerprint_at_turn(self.store, campaign_id, turn_id=...)
        deltas_replayed = await replay_to_turn(
            self.store, original=campaign_id, new=new_campaign_id, cutoff=...,
        )
        fp_new = await fingerprint_at_turn(self.store, new_campaign_id, turn_id=...)
        if fp_origin != fp_new:
            fingerprint_match = False
            # Fall back to copy-and-truncate
            await self._wipe_campaign(new_campaign_id)        # remove replayed rows
            await bulk_copy(self.store, original=campaign_id, new=new_campaign_id,
                            cutoff_turn_no=cutoff_turn_no)
            degraded = True

    # 4. Images
    img_result = await fork_image_files(src_dir, new_dir)
    await self.store.update_fork_provenance(
        new_campaign_id, image_handling=img_result.handling,
    )

    # 5. Done
    await self.events.emit_campaign_forked(
        campaign_id, new_campaign_id, fork_at_post_id, img_result.handling,
    )
    if make_active:
        await self.store.set_active_campaign(new_campaign_id)

    return ForkCampaignResult(
        new_campaign=await self.store.get_campaign(new_campaign_id),
        image_handling=img_result.handling,
        deltas_replayed=deltas_replayed,
        fingerprint_match=fingerprint_match,
        degraded=degraded,
        queued=False,
    )
```

Queueing: `_enqueue_fork` inserts a `pending_forks` row + returns `ForkCampaignResult(queued=True, ...)`. A coroutine on `Orchestrator` watches `_is_streaming` transitions and processes the queue on stream-completion.
